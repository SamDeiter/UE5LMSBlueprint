Implementation Code

// Determine if pin is connected
const isConnected = pin.links && pin.links.length > 0;

// EXEC PINS
if (pin.type === 'exec') {
    triangle.setAttribute('fill', isConnected ? 'white' : 'none');
    triangle.setAttribute('stroke', 'white');
}

// DATA PINS
const pinColor = Utils.getPinColor(pin.type);

// Circle (input or output)
circle.setAttribute('fill', isConnected ? pinColor : 'transparent');
circle.setAttribute('stroke', pinColor);

// Arrow (output only)
if (pin.dir === 'out') {
    arrow.setAttribute('fill', isConnected ? pinColor : 'transparent');
    arrow.setAttribute('stroke', isConnected ? 'none' : pinColor);
    arrow.setAttribute('stroke-width', isConnected ? '0' : '1');
}
Key Rule: ALL pins (exec and data) change from hollow → filled when connected!
